9. Button → Click hone par paragraph hide/show kare
<!DOCTYPE html>
<html>
<head>
  <title>Hide Show</title>
</head>
<body>
  <p id="para5">Main hide/show ho sakta hun.</p>
  <button onclick="toggleParagraph()">Hide/Show</button>

  <script>
    function toggleParagraph() {
      let para = document.getElementById("para5");
      if (para.style.display === "none") {
        para.style.display = "block";
      } else {
        para.style.display = "none";
      }
    }
  </script>
</body>
</html>

10. Button → Random background color set kare
<!DOCTYPE html>
<html>
<head>
  <title>Random Background</title>
</head>
<body>
  <button onclick="randomBackground()">Random Background</button>

  <script>
    function randomBackground() {
      let colors = ["red", "blue", "green", "yellow", "pink", "orange", "purple"];
      let random = colors[Math.floor(Math.random() * colors.length)];
      document.body.style.background = random;
    }
  </script>
</body>
</html>
